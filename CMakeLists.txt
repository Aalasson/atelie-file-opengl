cmake_minimum_required(VERSION 3.20)
project(AtelieFileOpenGL VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Esta primeira implementacao usa Win32/WGL. Consulte README.md.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
add_executable(AtelieFileOpenGL src/main.cpp src/game.cpp src/scene.cpp src/platform.cpp src/renderer.cpp)
target_include_directories(AtelieFileOpenGL PRIVATE src)
target_compile_definitions(AtelieFileOpenGL PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN ATELIE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(AtelieFileOpenGL PRIVATE opengl32 gdi32 user32 ole32 windowscodecs)
if(MSVC)
  target_compile_options(AtelieFileOpenGL PRIVATE /W4 /permissive- /utf-8 /MP)
endif()
add_custom_command(TARGET AtelieFileOpenGL POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
  "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:AtelieFileOpenGL>/assets")
enable_testing()
add_executable(atelie_core_tests tests/core_tests.cpp)
target_include_directories(atelie_core_tests PRIVATE src)
if(MSVC)
  target_compile_options(atelie_core_tests PRIVATE /W4 /permissive- /utf-8)
endif()
add_test(NAME core_logic COMMAND atelie_core_tests)
